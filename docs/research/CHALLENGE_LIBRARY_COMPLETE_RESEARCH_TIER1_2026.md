## PART 5: Data Model

```typescript
interface Challenge {
  // Identification
  id: string;
  slug: string;
  name: string;
  company: PropFirm;
  
  // Type
  type: 'free_competition' | 'paid_evaluation' | 'instant_funding';
  challengeType: '1-step' | '2-step' | 'competition' | 'instant';
  
  // Pricing
  isFree: boolean;
  entryFee: number | null;
  currency: 'USD' | 'EUR';
  refundable: boolean;
  refundConditions: string;
  
  // Account
  accountSize: number;
  scalingPotential: number;
  
  // Profit
  profitSplit: {
    initial: number;
    scaled?: number;
    maximum?: number;
  };
  
  // Rules
  rules: {
    profitTarget: number; // percentage
    maxDailyLoss: number; // percentage
    maxDrawdown: number; // percentage
    minTradingDays?: number;
    timeLimit?: number; // days, null if unlimited
    consistencyRule?: string;
  };
  
  // Payout
  payoutSpeed: 'instant' | 'same_day' | '24_hours' | 'weekly' | 'bi_weekly';
  firstPayoutDelay?: number; // days
  
  // Markets
  markets: ('forex' | 'futures' | 'indices' | 'commodities' | 'crypto')[];
  platforms: string[];
  
  // Status
  status: 'active' | 'upcoming' | 'ended';
  startDate?: Date;
  endDate?: Date;
  recurring: boolean;
  frequency?: 'monthly' | 'quarterly' | 'annual';
  
  // Metadata
  description: string;
  pros: string[];
  cons: string[];
  bestFor: string;
  officialUrl: string;
  logoUrl: string;
  
  // Analytics
  popularity: number; // 0-100
  successRate?: number; // percentage
  averagePassTime?: number; // days
}

interface PropFirm {
  id: string;
  name: string;
  slug: string;
  founded: number;
  totalPaid: number;
  reputation: number; // 0-100
  logoUrl: string;
  websiteUrl: string;
  description: string;
}
```
