### 🔄 Usage Examples

#### Basic Implementation
```typescript
import { usePushNotifications } from '@/hooks/usePushNotifications';

function NotificationSettings() {
  const {
    isSupported,
    isSubscribed,
    enableNotifications,
    sendTestNotification
  } = usePushNotifications();

  return (
    <div>
      {isSupported && (
        <button onClick={enableNotifications}>
          {isSubscribed ? 'Enabled' : 'Enable Notifications'}
        </button>
      )}
    </div>
  );
}
```

#### Dashboard Integration
```typescript
import { useDashboardNotifications } from '@/hooks/usePushNotifications';

function Dashboard() {
  const { sendDashboardNotification } = useDashboardNotifications();

  const handlePortfolioUpdate = async () => {
    await sendDashboardNotification(
      'Portfolio Updated',

      '/dashboard/portfolio'
    );
  };
}
```
